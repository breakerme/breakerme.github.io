AI学习之-格式占位符

# 整数类

占位符 含义 示例
%d 十进制有符号整数 42, -10
%i 同 %d（scanf可识别进制） 42
%u 十进制无符号整数 42
%o 八进制整数 52（即十进制42）
%x 十六进制整数（小写） 2a（即十进制42）
%X 十六进制整数（大写） 2A
int n = 42;
printf("%d  %o  %x  %X\n", n, n, n, n);
// 输出：42  52  2a  2A

# 浮点数类

占位符 含义 示例
%f 小数形式（默认6位小数） 3.141593
%e 科学计数法（小写） 3.141593e+00
%E 科学计数法（大写） 3.141593E+00
%g 自动选择 %f 或 %e 3.14159
double pi = 3.14159265;
printf("%f\n", pi);   // 3.141593
printf("%e\n", pi);   // 3.141593e+00
printf("%g\n", pi);   // 3.14159

# 字符类

占位符 含义 示例
%c 单个字符 A
%s 字符串 Hello
char ch = 'A';
char str[] = "Hello";
printf("%c\n", ch);   // A
printf("%s\n", str);  // Hello

# 指针 & 其他

占位符 含义 示例
%p 指针地址 0x7ffee4b2
%n 把已输出字符数写入变量 （特殊用途）
%% 输出一个 % 本身 %
int x = 42;
printf("%p\n", &x);   // 输出 x 的内存地址
printf("100%%\n");    // 输出 100%

# 宽度与精度控制

// 控制宽度
printf("%10d\n",  42);    //         42  （右对齐，占10位）
printf("%-10d|\n", 42);   // 42        |  （左对齐）
printf("%010d\n", 42);    // 0000000042  （用0填充）

// 控制小数位数
printf("%.2f\n",  3.14159);  // 3.14   （保留2位小数）
printf("%8.2f\n", 3.14159);  //     3.14（宽度8，2位小数）

// 控制字符串长度
printf("%.3s\n", "Hello");   // Hel    （只输出前3个字符）

修饰符 含义 示例
%ld long 整数 %ld
%lld long long 整数 %lld
%lf double 浮点数（scanf用） %lf
%hd short 整数 %hd
long long big = 9999999999LL;
